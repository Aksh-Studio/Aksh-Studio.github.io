rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Existing Dashboard Rules above...

    // --- Aksh Chat Rules ---

    // Spaces
    match /spaces/{spaceId} {
      // Anyone can read public spaces. Members can read private spaces.
      allow read: if resource.data.isPrivate == false || request.auth.uid in resource.data.members;
      
      // Any authenticated user can create a space
      allow create: if request.auth != null;
      
      // Only members can update the space
      allow update: if request.auth != null && request.auth.uid in resource.data.members;
      
      // Messages within Spaces
      match /messages/{messageId} {
        // Members (or anyone if public space) can read messages
        allow read: if get(/databases/$(database)/documents/spaces/$(spaceId)).data.isPrivate == false 
                    || request.auth.uid in get(/databases/$(database)/documents/spaces/$(spaceId)).data.members;
        
        // Members (or anyone if public space) can write
        allow create: if request.auth != null && 
                     (get(/databases/$(database)/documents/spaces/$(spaceId)).data.isPrivate == false 
                      || request.auth.uid in get(/databases/$(database)/documents/spaces/$(spaceId)).data.members)
                     && request.resource.data.senderId == request.auth.uid;

        // Authors can update/delete their own messages
        allow update, delete: if request.auth != null && resource.data.senderId == request.auth.uid;
        
        // Threads (Replies)
        match /replies/{replyId} {
          allow read: if get(/databases/$(database)/documents/spaces/$(spaceId)).data.isPrivate == false 
                      || request.auth.uid in get(/databases/$(database)/documents/spaces/$(spaceId)).data.members;
          
          allow create: if request.auth != null && 
                       (get(/databases/$(database)/documents/spaces/$(spaceId)).data.isPrivate == false 
                        || request.auth.uid in get(/databases/$(database)/documents/spaces/$(spaceId)).data.members)
                       && request.resource.data.senderId == request.auth.uid;
                       
          allow update, delete: if request.auth != null && resource.data.senderId == request.auth.uid;
        }
      }
    }

    // Direct Messages
    match /direct_chats/{chatId} {
      // Only participants can read or update
      allow read, update: if request.auth != null && request.auth.uid in resource.data.participants;
      
      // Authenticated users can create DM threads if they are a participant
      allow create: if request.auth != null && request.auth.uid in request.resource.data.participants;

      match /messages/{messageId} {
        // Only chat participants can read or write
        allow read: if request.auth != null && request.auth.uid in get(/databases/$(database)/documents/direct_chats/$(chatId)).data.participants;
        
        allow create: if request.auth != null && 
                      request.auth.uid in get(/databases/$(database)/documents/direct_chats/$(chatId)).data.participants &&
                      request.resource.data.senderId == request.auth.uid;

        allow update, delete: if request.auth != null && resource.data.senderId == request.auth.uid;
      }
    }
  }
}
